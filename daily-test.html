async function finishTest(cls) { const prefix = cls === '9' ? 'p2' : 'p3'; let s = state[cls]; stopTimer(cls); playVictorySound(); let correct = 0; const chapters = syllabus[cls]; const byChapter = chapters.map(() => ({correct: 0, total: 0})); s.questions.forEach((q, i) => { byChapter[q.chapter].total++; if(s.answers[i] === q.answer) { correct++; byChapter[q.chapter].correct++; } }); const totalQ = s.questions.length; const percent = totalQ > 0 ? Math.round((correct / totalQ) * 100) : 0; // समय की गणना (Time Calculation) let endTime = new Date(); let timeTakenSec = s.startTime ? Math.round((endTime - s.startTime) / 1000) : 0; if(timeTakenSec < 1) timeTakenSec = 1; let mins = Math.floor(timeTakenSec / 60); let secs = timeTakenSec % 60; let timeStr = mins > 0 ? `${mins} मिनट ${secs} सेकंड` : `${secs} सेकंड`; // डेटाबेस में रिजल्ट सेव करने का कोड try { let studentName = localStorage.getItem("studentName") || "विद्यार्थी"; let studentId = localStorage.getItem("studentId") || "—"; const payload = { student_name: studentName, student_id: studentId, class_name: "कक्षा " + cls + "वीं", test_type: s.testMode === 'range' ? "प्रगतिशील पाठ्यक्रम टेस्ट" : `अध्याय टेस्ट (${syllabus[cls][s.upto - 1]})`, correct_answers: correct, total_questions: totalQ, percentage: percent, time_taken: timeStr, created_at: new Date().toISOString() }; console.log("Supabase में डेटा भेजा जा रहा है:", payload); let { data, error } = await supabaseClient.from("student_test_results").insert([payload]); if (error) { console.error("Supabase Save Error:", error.message); alert("परिणाम डेटाबेस में सेव होने में त्रुटि: " + error.message); } else { console.log("परिणाम सफलतापूर्वक सेव हो गया!", data); } } catch(err) { console.error("Network or Code Error:", err); } document.getElementById(`${prefix}-scoreText`).textContent = `${correct} / ${totalQ}`; document.getElementById(`${prefix}-percentText`).textContent = `${percent}% अंक प्राप्त हुए (${timeStr})`; let resultsHTML = ""; byChapter.forEach((chData, idx) => { if(chData.total > 0) { let chPercent = Math.round((chData.correct / chData.total) * 100); resultsHTML += `
अध्याय ${idx + 1}: ${chapters[idx]} ${chData.correct}/${chData.total} (${chPercent}%)
`; } }); document.getElementById(`${prefix}-chapterResults`).innerHTML = resultsHTML; document.getElementById(`${prefix}-testSec`).style.display = "none"; document.getElementById(`${prefix}-resultSec`).style.display = "flex"; }
